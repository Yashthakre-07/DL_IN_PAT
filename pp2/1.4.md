# Step 1.4: Frontend Analytical Comparator UI (Code Implementation)

## Overview
This step implements the React frontend for the Comparator using Next.js, shadcn/ui, and Tailwind.

## 1. UI Component (`stich/app/comparator/page.tsx`)
```tsx
"use client";
import { useState } from 'react';
import axios from 'axios';
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";

export default function ComparatorPage() {
  const [testFile, setTestFile] = useState<File | null>(null);
  const [refFile, setRefFile] = useState<File | null>(null);
  const [results, setResults] = useState<any>(null);
  const [loading, setLoading] = useState(false);

  const handleRun = async () => {
    if (!testFile) return alert("Test image required");
    setLoading(true);
    
    const formData = new FormData();
    formData.append("test_image", testFile);
    if (refFile) formData.append("reference_image", refFile);

    try {
      const res = await axios.post("http://localhost:8000/comparator/evaluate", formData);
      setResults(res.data);
    } catch (err) {
      console.error(err);
      alert("Error evaluating metrics");
    }
    setLoading(false);
  };

  return (
    <div className="p-8 max-w-6xl mx-auto">
      <h1 className="text-3xl font-bold mb-8">Analytical Comparator</h1>
      
      <div className="grid grid-cols-2 gap-8 mb-8">
        <Card>
          <CardHeader><CardTitle>Test Image (Predicted/Reconstructed)</CardTitle></CardHeader>
          <CardContent>
            <Input type="file" onChange={e => setTestFile(e.target.files?.[0] || null)} />
          </CardContent>
        </Card>
        <Card>
          <CardHeader><CardTitle>Reference Image (Ground Truth)</CardTitle></CardHeader>
          <CardContent>
            <Input type="file" onChange={e => setRefFile(e.target.files?.[0] || null)} />
          </CardContent>
        </Card>
      </div>

      <Button onClick={handleRun} disabled={loading} className="w-full mb-8">
        {loading ? "Analyzing..." : "Run Analysis"}
      </Button>

      {results && (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
          {/* Primary Metrics (S3IM, GMSD, SSIM, etc) */}
          <Card className="border-blue-500 border-2">
            <CardHeader><CardTitle className="text-blue-600">Primary Structural Metrics</CardTitle></CardHeader>
            <CardContent>
              {Object.entries(results.metrics.Primary).map(([k, v]) => (
                <div key={k} className="flex justify-between py-2 border-b">
                  <span className="font-bold">{k}</span>
                  <span>{Number(v).toFixed(4)}</span>
                </div>
              ))}
            </CardContent>
          </Card>

          {/* Difference Heatmap */}
          {results.heatmap_base64 && (
            <Card>
              <CardHeader><CardTitle>Difference Heatmap</CardTitle></CardHeader>
              <CardContent className="flex justify-center">
                <img src={`data:image/png;base64,${results.heatmap_base64}`} alt="Heatmap" className="rounded-lg shadow-lg" />
              </CardContent>
            </Card>
          )}
        </div>
      )}
    </div>
  );
}
```
