# Step 2.2: Celery Training Job System (Code Implementation)

## Overview
This creates the background worker to train models (PAQNet, EfficientNet) using Huber Loss/MAE, streaming states to Redis.

## 1. Train Loop (`app/ml/training/train_loop.py`)
```python
import torch
import torch.nn as nn
from app.services.inference_service import InferenceService

def run_training(epochs, model_name, train_loader, val_loader, task_instance):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    
    # Load architecture
    infer_service = InferenceService()
    infer_service.load_model(model_name, "") # Load empty structure
    model = infer_service.active_model.to(device)
    
    criterion = nn.HuberLoss() # Paper used Huber and evaluated MAE
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)
    
    best_val_loss = float('inf')
    
    for epoch in range(epochs):
        model.train()
        train_loss = 0
        for images, labels in train_loader:
            images, labels = images.to(device), labels.to(device)
            optimizer.zero_grad()
            outputs = model(images)
            loss = criterion(outputs.squeeze(), labels)
            loss.backward()
            optimizer.step()
            train_loss += loss.item()
            
        train_loss /= len(train_loader)
        
        # Validation
        model.eval()
        val_loss = 0
        with torch.no_grad():
            for images, labels in val_loader:
                images, labels = images.to(device), labels.to(device)
                outputs = model(images)
                loss = criterion(outputs.squeeze(), labels)
                val_loss += loss.item()
                
        val_loss /= len(val_loader)
        
        if val_loss < best_val_loss:
            best_val_loss = val_loss
            torch.save(model.state_dict(), f"../saved_models/best_{model_name}.pth")
            
        # Broadcast via Celery
        if task_instance:
            task_instance.update_state(state='TRAINING', meta={
                'epoch': epoch + 1,
                'train_loss': train_loss,
                'val_loss': val_loss
            })
            
    return {"status": "Complete", "best_val": best_val_loss}
```

## 2. Celery Worker (`app/workers/celery_app.py`)
```python
from celery import Celery
from app.ml.training.train_loop import run_training

celery_app = Celery("training_worker", broker="redis://localhost:6379/0", backend="redis://localhost:6379/0")

@celery_app.task(bind=True)
def start_training_task(self, config):
    # Dummy loaders for illustration
    train_loader = [] 
    val_loader = []
    return run_training(config['epochs'], config['model_name'], train_loader, val_loader, self)
```
