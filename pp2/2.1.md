# Step 2.1: Dataset Preprocessing & Ingestion Service (Code Implementation)

## Overview
This step implements the `dataset_service.py` to ingest zip files containing training images and perform the 70/15/15 splits, outputting ready PyTorch DataLoaders.

## 1. Dataset Service (`app/services/dataset_service.py`)
```python
import os
import zipfile
import pandas as pd
from torch.utils.data import Dataset, DataLoader
from PIL import Image
from torchvision import transforms

class PhotoacousticDataset(Dataset):
    def __init__(self, image_paths, labels, transform=None):
        self.image_paths = image_paths
        self.labels = labels
        self.transform = transform

    def __len__(self):
        return len(self.image_paths)

    def __getitem__(self, idx):
        img_path = self.image_paths[idx]
        image = Image.open(img_path).convert("L") # Grayscale
        if self.transform:
            image = self.transform(image)
        label = self.labels[idx]
        return image, torch.tensor(label, dtype=torch.float32)

class DatasetService:
    def __init__(self, upload_dir="../data/temp"):
        self.upload_dir = upload_dir
        os.makedirs(self.upload_dir, exist_ok=True)
        self.transform = transforms.Compose([
            transforms.Resize((128, 128)),
            transforms.ToTensor()
        ])

    def process_zip(self, zip_path: str):
        extract_dir = os.path.join(self.upload_dir, "extracted")
        with zipfile.ZipFile(zip_path, 'r') as zip_ref:
            zip_ref.extractall(extract_dir)
            
        # Example logic: Read a metadata.csv inside the zip
        csv_path = os.path.join(extract_dir, "metadata.csv")
        if not os.path.exists(csv_path):
            raise ValueError("metadata.csv missing in dataset zip.")
            
        df = pd.read_csv(csv_path)
        
        # 70/15/15 split logic (from paper)
        from sklearn.model_selection import train_test_split
        train_df, temp_df = train_test_split(df, test_size=0.3, random_state=42)
        val_df, test_df = train_test_split(temp_df, test_size=0.5, random_state=42)
        
        return train_df, val_df, test_df

    def create_dataloaders(self, train_df, val_df, test_df, batch_size=16):
        train_ds = PhotoacousticDataset(train_df['image_path'].values, train_df['score'].values, self.transform)
        val_ds = PhotoacousticDataset(val_df['image_path'].values, val_df['score'].values, self.transform)
        
        return {
            "train": DataLoader(train_ds, batch_size=batch_size, shuffle=True),
            "val": DataLoader(val_ds, batch_size=batch_size, shuffle=False)
        }
```
