# Step 2.4: AI Training Studio UI (Code Implementation)

## 1. Training Dashboard (`stich/app/training/page.tsx`)
```tsx
"use client";
import { useState, useEffect } from "react";
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, Legend } from 'recharts';
import { Card, CardHeader, CardTitle, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";

export default function TrainingStudio() {
    const [data, setData] = useState<any[]>([]);
    const [taskId, setTaskId] = useState<string | null>(null);

    const startTraining = async () => {
        // Trigger celery task via REST
        const res = await fetch("http://localhost:8000/api/v1/training/start", { method: "POST" });
        const json = await res.json();
        setTaskId(json.task_id);
    };

    useEffect(() => {
        if (!taskId) return;
        const ws = new WebSocket(`ws://localhost:8000/api/v1/training/ws/${taskId}`);
        ws.onmessage = (event) => {
            const msg = JSON.parse(event.data);
            if (msg.epoch) {
                setData(prev => [...prev, { epoch: msg.epoch, TrainLoss: msg.train_loss, ValLoss: msg.val_loss }]);
            }
        };
        return () => ws.close();
    }, [taskId]);

    return (
        <div className="p-8">
            <h1 className="text-3xl font-bold mb-4">AI Training Studio</h1>
            <Button onClick={startTraining} className="mb-8">Start Training PAQNet</Button>
            
            <Card>
                <CardHeader><CardTitle>Live Loss Curve</CardTitle></CardHeader>
                <CardContent>
                    <LineChart width={800} height={400} data={data}>
                        <CartesianGrid strokeDasharray="3 3" />
                        <XAxis dataKey="epoch" />
                        <YAxis />
                        <Tooltip />
                        <Legend />
                        <Line type="monotone" dataKey="TrainLoss" stroke="#8884d8" />
                        <Line type="monotone" dataKey="ValLoss" stroke="#82ca9d" />
                    </LineChart>
                </CardContent>
            </Card>
        </div>
    );
}
```
