# Step 1.3: Analytical Comparator Endpoint (Code Implementation)

## Overview
This creates the actual API route `POST /api/v1/comparator/evaluate` that receives the images from the frontend and returns the evaluated metrics.

## 1. Pydantic Schemas (`app/schemas/comparator.py`)
```python
from pydantic import BaseModel
from typing import Dict

class ComparatorResponse(BaseModel):
    metrics: Dict[str, Dict[str, float]]
    heatmap_base64: str = None
    status: str
```

## 2. Comparator Router (`app/api/routes/comparator.py`)
```python
from fastapi import APIRouter, UploadFile, File, HTTPException
from app.services.metric_service import MetricService
from app.schemas.comparator import ComparatorResponse
import numpy as np
import cv2
import base64

router = APIRouter()

def process_upload(file: UploadFile) -> np.ndarray:
    contents = file.file.read()
    nparr = np.frombuffer(contents, np.uint8)
    img = cv2.imdecode(nparr, cv2.IMREAD_GRAYSCALE)
    if img is None:
        raise ValueError("Invalid image file")
    # Resize to standard size (e.g., 128x128)
    img = cv2.resize(img, (128, 128))
    # Normalize to [0,1]
    return img.astype(np.float32) / 255.0

@router.post("/evaluate", response_model=ComparatorResponse)
async def evaluate_comparator(
    test_image: UploadFile = File(...),
    reference_image: UploadFile = File(None)
):
    try:
        test_np = process_upload(test_image)
        ref_np = process_upload(reference_image) if reference_image else None
        
        # 1. Run Metrics
        metrics = MetricService.run_all_metrics(test_np, ref_np)
        
        # 2. Generate Difference Heatmap (if ref is provided)
        heatmap_b64 = None
        if ref_np is not None:
            diff = np.abs(ref_np - test_np)
            diff_mapped = cv2.applyColorMap((diff * 255).astype(np.uint8), cv2.COLORMAP_JET)
            _, buffer = cv2.imencode('.png', diff_mapped)
            heatmap_b64 = base64.b64encode(buffer).decode('utf-8')
            
        return ComparatorResponse(
            metrics=metrics,
            heatmap_base64=heatmap_b64,
            status="success"
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```

## 3. Register Router in `main.py`
Add this to `app/main.py`:
```python
from app.api.routes import comparator
app.include_router(comparator.router, prefix="/comparator", tags=["Analytical Comparator"])
```
